# Nur Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Nur, a zero-config script discovery and execution engine that finds tasks across multiple ecosystems (npm, make, pdm, poe, just, Taskfile) in the current directory and runs them via a TUI picker or direct CLI invocation.

**Architecture:** A small core orchestrates one **Provider** per source format. Each provider detects its source file and produces `Task` objects. A `Registry` aggregates tasks and resolves names (bare or `prefix:name`). The CLI dispatches either to a direct-run path (inherited stdio) or to a lazily-imported Textual three-pane TUI. Every provider is built around a **pure parse function** so it is fully unit-testable without the underlying tool installed.

**Tech Stack:** Python ≥ 3.11 (stdlib `tomllib`), Textual (TUI, lazy-imported), `uv` (env/build), `ruff` (lint/format), `pytest` + `pytest-cov` (tests). Hatchling build backend, `src/` layout.

## Global Constraints

Every task's requirements implicitly include this section.

- **Python floor:** ≥ 3.11 (required for stdlib `tomllib`). `requires-python = ">=3.11"`.
- **Runtime dependencies:** `textual` only. Everything else uses the stdlib (`json`, `tomllib`, `subprocess`, `difflib`, `re`, `pathlib`, `logging`). No YAML library — Taskfile is read via tool delegation, never parsed directly.
- **Providers never raise from `discover()`:** on a malformed/absent source they log a warning via `logging.getLogger("nur")` and return `[]`. The discovery orchestrator adds a backstop `try/except` regardless.
- **No shell wrapping:** tasks are executed by passing an argv list directly to `subprocess` (no `shell=True`), to avoid quoting surprises.
- **Textual is lazy-imported:** `import` of anything under `nur.tui` happens only inside the function that launches the picker, never at module top level of `nur.cli` or the core.
- **Discovery scope:** current working directory only. Never walk up or recurse.
- **Coverage floor:** `--cov-fail-under=90` in pytest config. Use `# pragma: no cover` only for genuinely unreachable lines (e.g. the `if __name__` guard).
- **Commits:** Conventional Commits (`feat:`, `test:`, `chore:`, `docs:`). No AI-attribution trailers or co-author lines.
- **Task addressing:** qualified name is always `prefix:name` (e.g. `npm:test`).

```python
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Protocol, runtime_checkable


@dataclass(frozen=True, slots=True)
class Task:
    name: str
    prefix: str
    argv_base: tuple[str, ...]
    description: str | None = None
    definition: str = ""
    source_file: str = ""

    @property
    def qualified_name(self) -> str:
        return f"{self.prefix}:{self.name}"

    def run_argv(self, extra_args: list[str] | None = None) -> list[str]:
        return [*self.argv_base, *(extra_args or [])]


@runtime_checkable
class Provider(Protocol):
    prefix: str

    def detect(self, cwd: Path) -> bool: ...

    def discover(self, cwd: Path) -> list[Task]: ...
```

```python
from __future__ import annotations

import difflib

from .models import Task


class ResolutionError(Exception):
    """Base class for name-resolution failures."""


class UnknownTaskError(ResolutionError):
    def __init__(self, query: str, suggestions: list[str]) -> None:
        self.query = query
        self.suggestions = suggestions
        super().__init__(f"unknown task: {query!r}")


class AmbiguousTaskError(ResolutionError):
    def __init__(self, name: str, candidates: list[str]) -> None:
        self.name = name
        self.candidates = candidates
        super().__init__(f"ambiguous task: {name!r}")


class Registry:
    def __init__(self, tasks: list[Task]) -> None:
        self._tasks = tasks

    def is_empty(self) -> bool:
        return not self._tasks

    def all(self) -> list[Task]:
        return list(self._tasks)

    def groups(self) -> dict[str, list[Task]]:
        out: dict[str, list[Task]] = {}
        for t in self._tasks:
            out.setdefault(t.prefix, []).append(t)
        return out

    def _suggest(self, query: str) -> list[str]:
        pool = [t.name for t in self._tasks] + [t.qualified_name for t in self._tasks]
        return difflib.get_close_matches(query, pool, n=3, cutoff=0.5)

    def resolve(self, query: str) -> Task:
        if ":" in query:
            prefix, _, name = query.partition(":")
            for t in self._tasks:
                if t.prefix == prefix and t.name == name:
                    return t
            raise UnknownTaskError(query, self._suggest(query))
        matches = [t for t in self._tasks if t.name == query]
        if len(matches) == 1:
            return matches[0]
        if not matches:
            raise UnknownTaskError(query, self._suggest(query))
        raise AmbiguousTaskError(query, [t.qualified_name for t in matches])
```

```python
from __future__ import annotations

import json
import logging
from pathlib import Path

from nur.models import Task

log = logging.getLogger("nur")

# (lockfile, package manager) in precedence order.
_LOCKFILES: tuple[tuple[str, str], ...] = (
    ("bun.lockb", "bun"),
    ("pnpm-lock.yaml", "pnpm"),
    ("yarn.lock", "yarn"),
    ("package-lock.json", "npm"),
)


def resolve_pm(cwd: Path) -> str:
    for lockfile, pm in _LOCKFILES:
        if (cwd / lockfile).exists():
            return pm
    return "npm"


class NpmProvider:
    prefix = "npm"

    def detect(self, cwd: Path) -> bool:
        return (cwd / "package.json").is_file()

    def discover(self, cwd: Path) -> list[Task]:
        try:
            data = json.loads((cwd / "package.json").read_text())
        except (OSError, json.JSONDecodeError) as exc:
            log.warning("nur: skipping package.json (%s)", exc)
            return []
        scripts = data.get("scripts") or {}
        pm = resolve_pm(cwd)
        return [
            Task(
                name=name,
                prefix=self.prefix,
                argv_base=(pm, "run", name),
                definition=str(command),
                source_file="package.json",
            )
            for name, command in scripts.items()
        ]
```

```python
from __future__ import annotations

import logging
import tomllib
from pathlib import Path

from nur.models import Task

log = logging.getLogger("nur")


def _load_scripts(cwd: Path) -> dict | None:
    """Return the [tool.pdm.scripts] table, or None if absent/unreadable."""
    path = cwd / "pyproject.toml"
    try:
        data = tomllib.loads(path.read_text())
    except (OSError, tomllib.TOMLDecodeError) as exc:
        log.warning("nur: skipping pyproject.toml (%s)", exc)
        return None
    return data.get("tool", {}).get("pdm", {}).get("scripts")


def _definition_and_help(value: object) -> tuple[str, str | None]:
    if isinstance(value, dict):
        body = value.get("cmd") or value.get("shell") or value.get("call") or ""
        if isinstance(body, list):
            body = " ".join(str(x) for x in body)
        return str(body), value.get("help")
    return str(value), None


class PdmProvider:
    prefix = "pdm"

    def detect(self, cwd: Path) -> bool:
        scripts = _load_scripts(cwd)
        return bool(scripts)

    def discover(self, cwd: Path) -> list[Task]:
        scripts = _load_scripts(cwd)
        if not scripts:
            return []
        tasks: list[Task] = []
        for name, value in scripts.items():
            if name == "_":  # PDM reserved global-options key
                continue
            definition, help_text = _definition_and_help(value)
            tasks.append(
                Task(
                    name=name,
                    prefix=self.prefix,
                    argv_base=("pdm", "run", name),
                    description=help_text,
                    definition=definition,
                    source_file="pyproject.toml",
                )
            )
        return tasks
```

```python
from nur.models import Task

log = logging.getLogger("nur")


def _load_tasks(cwd: Path) -> dict | None:
    path = cwd / "pyproject.toml"
    try:
        data = tomllib.loads(path.read_text())
    except (OSError, tomllib.TOMLDecodeError) as exc:
        log.warning("nur: skipping pyproject.toml (%s)", exc)
        return None
    return data.get("tool", {}).get("poe", {}).get("tasks")


def _definition_and_help(value: object) -> tuple[str, str | None]:
    if isinstance(value, dict):
        body = value.get("cmd") or value.get("shell") or value.get("script") or ""
        if isinstance(body, list):
            body = " ".join(str(x) for x in body)
        return str(body), value.get("help")
    return str(value), None


class PoeProvider:
    prefix = "poe"

    def detect(self, cwd: Path) -> bool:
        return bool(_load_tasks(cwd))

    def discover(self, cwd: Path) -> list[Task]:
        tasks_table = _load_tasks(cwd)
        if not tasks_table:
            return []
        tasks: list[Task] = []
        for name, value in tasks_table.items():
            definition, help_text = _definition_and_help(value)
            tasks.append(
                Task(
                    name=name,
                    prefix=self.prefix,
                    argv_base=("poe", name),
                    description=help_text,
                    definition=definition,
                    source_file="pyproject.toml",
                )
            )
        return tasks
```

```python
from __future__ import annotations

import logging
import re
import subprocess
from pathlib import Path

from nur.models import Task

log = logging.getLogger("nur")

# A target definition line: "name:" or "name: prereqs", not a variable assignment.
_TARGET_RE = re.compile(r"^([A-Za-z0-9][^\s:=%#]*)\s*:(?!=)")
# A described target: "name: ... ## description".
_DESC_RE = re.compile(r"^([A-Za-z0-9][A-Za-z0-9._-]*)\s*:.*?##\s*(.*)$")


def parse_database(output: str) -> list[str]:
    names: list[str] = []
    seen: set[str] = set()
    lines = output.splitlines()
    for i, line in enumerate(lines):
        match = _TARGET_RE.match(line)
        if not match:
            continue
        name = match.group(1).strip()
        if i > 0 and lines[i - 1].strip() == "# Not a target:":
            continue
        if name.startswith(".") or "%" in name or name == "Makefile":
            continue
        if name in seen:
            continue
        seen.add(name)
        names.append(name)
    return names


def parse_descriptions(text: str) -> dict[str, str]:
    out: dict[str, str] = {}
    for line in text.splitlines():
        match = _DESC_RE.match(line)
        if match:
            out[match.group(1)] = match.group(2).strip()
    return out


class MakeProvider:
    prefix = "make"

    def detect(self, cwd: Path) -> bool:
        return (cwd / "Makefile").is_file()

    def discover(self, cwd: Path) -> list[Task]:
        try:
            result = subprocess.run(
                ["make", "-pRrq"],
                cwd=cwd,
                capture_output=True,
                text=True,
            )
        except FileNotFoundError:
            log.warning("nur: 'make' not installed; skipping Makefile tasks")
            return []
        names = parse_database(result.stdout)
        try:
            descriptions = parse_descriptions((cwd / "Makefile").read_text())
        except OSError:
            descriptions = {}
        return [
            Task(
                name=name,
                prefix=self.prefix,
                argv_base=("make", name),
                description=descriptions.get(name),
                source_file="Makefile",
            )
            for name in names
        ]
```

```python
from pathlib import Path

from .models import Task
from .providers import PROVIDERS
from .registry import Registry

log = logging.getLogger("nur")


def discover(cwd: Path, providers: list | None = None) -> Registry:
    providers = PROVIDERS if providers is None else providers
    tasks: list[Task] = []
    for provider in providers:
        try:
            if provider.detect(cwd):
                tasks.extend(provider.discover(cwd))
        except Exception as exc:  # backstop: a provider must never break discovery

    return Registry(tasks)
```

```python
from __future__ import annotations

import signal
import subprocess
from collections.abc import Callable
from pathlib import Path

from .models import Task


def run_direct(task: Task, extra_args: list[str], cwd: Path) -> int:
    """Run a task with inherited stdio; return the child's exit code."""
    completed = subprocess.run(task.run_argv(extra_args), cwd=cwd)
    return completed.returncode


class ProcessRunner:
    """Runs a command with combined output captured and streamed line-by-line.

    Blocking; intended to be driven from a background thread by the TUI.
    """

    def __init__(self) -> None:
        self._proc: subprocess.Popen | None = None

    def run(
        self,
        argv: list[str],
        cwd: Path,
        on_line: Callable[[str], None],
    ) -> int:
        self._proc = subprocess.Popen(
            argv,
            cwd=cwd,
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
            stdin=subprocess.DEVNULL,
            text=True,
            bufsize=1,
        )
        assert self._proc.stdout is not None
        for line in self._proc.stdout:
            on_line(line)
        self._proc.stdout.close()
        code = self._proc.wait()
        self._proc = None
        return code

    def interrupt(self) -> None:
        if self._proc is not None and self._proc.poll() is None:
            self._proc.send_signal(signal.SIGINT)
```

```python
from __future__ import annotations

import argparse
import sys
from pathlib import Path

import nur
from nur.discovery import discover
from nur.execution import run_direct
from nur.registry import AmbiguousTaskError, Registry, UnknownTaskError


def split_passthrough(argv: list[str]) -> tuple[list[str], list[str]]:
    if "--" in argv:
        i = argv.index("--")
        return argv[:i], argv[i + 1 :]
    return list(argv), []


def format_list(registry: Registry) -> str:
    lines: list[str] = []
    for prefix, tasks in registry.groups().items():
        lines.append(prefix)
        for task in tasks:
            desc = f"  {task.description}" if task.description else ""
            lines.append(f"  {task.name}{desc}")
    return "\n".join(lines)


def _build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="nur", description="Script discovery and execution engine."
    )
    # Manual version flag (not action="version") so main() can return 0 rather
    # than raising SystemExit — keeps main()'s int contract clean and testable.
    parser.add_argument("-V", "--version", action="store_true", help="print version and exit")
    parser.add_argument(
        "task", nargs="?", help="task to run, e.g. 'test' or 'make:test'; 'list' to list"
    )
    return parser


def main(argv: list[str] | None = None) -> int:
    raw = sys.argv[1:] if argv is None else argv
    left, extra = split_passthrough(raw)
    parser = _build_parser()
    args = parser.parse_args(left)
    cwd = Path.cwd()

    if args.version:
        print(nur.__version__)
        return 0

    registry = discover(cwd)

    if args.task == "list":
        print(format_list(registry))
        return 0

    if args.task is None:
        if registry.is_empty():
            print("nur: no supported task files found in this directory.")
            return 0
        from nur.tui.app import launch  # lazy import — keeps direct path fast

        return launch(registry, cwd)

    try:
        task = registry.resolve(args.task)
    except AmbiguousTaskError as exc:
        print(
            f"nur: '{exc.name}' is ambiguous; candidates: {', '.join(exc.candidates)}",
            file=sys.stderr,
        )
        return 2
    except UnknownTaskError as exc:
        msg = f"nur: unknown task '{exc.query}'"
        if exc.suggestions:
            msg += f"; did you mean: {', '.join(exc.suggestions)}?"
        print(msg, file=sys.stderr)
        return 2

    return run_direct(task, extra, cwd)
```

```python
from __future__ import annotations

from pathlib import Path

from textual.app import App, ComposeResult
from textual.containers import Horizontal, Vertical
from textual.widgets import Input, ListItem, ListView, RichLog, Static

from nur.models import Task
from nur.registry import Registry


class TaskItem(ListItem):
    def __init__(self, task: Task) -> None:
        super().__init__(Static(f"  {task.name}"))
        self.task = task


class HeaderItem(ListItem):
    def __init__(self, prefix: str) -> None:
        super().__init__(Static(prefix, classes="group-header"))
        self.disabled = True
        self.task = None


class NurApp(App):
    CSS_PATH = "app.tcss"
    BINDINGS = [
        ("q", "quit", "quit"),
        ("escape", "quit", "quit"),
        ("slash", "focus_filter", "filter"),
        ("j", "next", "down"),
        ("k", "prev", "up"),
        ("r", "run", "run"),
        ("question_mark", "toggle_help", "help"),
    ]

    def __init__(self, registry: Registry, cwd: Path) -> None:
        super().__init__()
        self._registry = registry
        self._cwd = cwd
        self.selected_task: Task | None = None
        self.exit_code = 0

    def compose(self) -> ComposeResult:
        yield Input(placeholder="filter", id="filter")
        with Horizontal():
            yield ListView(id="tasks")
            with Vertical(id="right"):
                yield Static("", id="detail")
                yield RichLog(id="output", highlight=False, markup=False)
                yield Static("", id="status")

    def on_mount(self) -> None:
        self._rebuild("")
        # Focus the list (not the Input) so '/', 'j', 'k', 'r', 'q' fire as
        # app bindings instead of being typed into the filter box.
        self.query_one("#tasks", ListView).focus()

    def _matches(self, query: str) -> list[Task]:
        q = query.lower()
        out = []
        for t in self._registry.all():
            hay = f"{t.name} {t.description or ''}".lower()
            if q in hay:
                out.append(t)
        return out

    def _rebuild(self, query: str) -> None:
        lv = self.query_one("#tasks", ListView)
        lv.clear()
        matches = self._matches(query)
        current_prefix = None
        first_index = None  # index of the first selectable TaskItem
        idx = 0
        for task in matches:
            if task.prefix != current_prefix:
                current_prefix = task.prefix
                lv.append(HeaderItem(task.prefix))
                idx += 1
            lv.append(TaskItem(task))
            if first_index is None:
                first_index = idx
            idx += 1
        if matches:
            lv.index = first_index
            self._select_task(matches[0])  # drive detail from data, not widgets
        else:
            self.selected_task = None
            self.query_one("#detail", Static).update("no matches")

    def _select_task(self, task: Task) -> None:
        self.selected_task = task
        detail = (
            f"{task.qualified_name}\n"
            f"{task.description or ''}\n\n"
            f"$ {' '.join(task.argv_base)}\n"
            f"from {task.source_file}"
        )
        if task.definition:
            detail += f"\n\n{task.definition}"
        self.query_one("#detail", Static).update(detail)

    def on_list_view_highlighted(self, event: ListView.Highlighted) -> None:
        task = getattr(event.item, "task", None)
        if task is not None:
            self._select_task(task)

    def on_list_view_selected(self, event: ListView.Selected) -> None:
        # Enter on a task row runs it (Task 14 wires the execution).
        self.action_run()

    def on_input_changed(self, event: Input.Changed) -> None:
        self._rebuild(event.value)

    def action_focus_filter(self) -> None:
        self.query_one("#filter", Input).focus()

    def action_next(self) -> None:
        self.query_one("#tasks", ListView).action_cursor_down()

    def action_prev(self) -> None:
        self.query_one("#tasks", ListView).action_cursor_up()

    def action_toggle_help(self) -> None:
        self.query_one("#status", Static).update("keys: j/k move · / filter · r run · q quit")

    def action_run(self) -> None:
        # Execution wiring added in Task 14.
        if self.selected_task is not None:
            self.query_one("#status", Static).update(
                f"would run: {self.selected_task.qualified_name}"
            )


def launch(registry: Registry, cwd: Path) -> int:  # pragma: no cover
    app = NurApp(registry, cwd)
    app.run()  # needs a real terminal; exercised manually, not in CI
    return app.exit_code
```

`launch()` carries `# pragma: no cover` because `app.run()` requires a real
terminal and cannot run under pytest (the TUI is tested via `App.run_test()`
against `NurApp` directly). This keeps the 90% coverage floor honest.

Notes:
- The `/` binding focuses the Input; subsequent characters are typed into it because the Input has focus, and `on_input_changed` rebuilds the list live. Do **not** bind `Tab` — Textual's built-in focus traversal already cycles focus across the focusable widgets (`#filter`, `#tasks`, `#output`), which satisfies the spec's "Tab cycles panes".
- Static widgets (`#detail`, `#status`) are non-focusable, so Tab skips them.

```python
def action_run(self) -> None:
    if self._running or self.selected_task is None:
        return
    task = self.selected_task
    self._running = True
    self._output_lines = []
    self.query_one("#output", RichLog).clear()
    self.query_one("#status", Static).update(f"running… $ {' '.join(task.argv_base)}")
    self.run_worker(
        lambda: self._runner.run(
            task.run_argv(),
            self._cwd,
            lambda line: self.call_from_thread(self._append_output, line),
        ),
        thread=True,
        exclusive=True,
        name="task-run",
    )


def _append_output(self, line: str) -> None:
    text = line.rstrip("\n")
    self._output_lines.append(text)
    self.query_one("#output", RichLog).write(text)


def on_worker_state_changed(self, event) -> None:
    from textual.worker import WorkerState

    if event.worker.name == "task-run" and event.worker.state in (
        WorkerState.SUCCESS,
        WorkerState.ERROR,
    ):
        code = event.worker.result if event.worker.state == WorkerState.SUCCESS else 1
        self.exit_code = code if isinstance(code, int) else 1
        self._running = False
        self.query_one("#status", Static).update(f"exited({self.exit_code})")


def action_interrupt(self) -> None:
    self._runner.interrupt()
```
